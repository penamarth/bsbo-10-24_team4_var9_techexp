@startuml
skinparam sequenceMessageAlign center
skinparam maxMessageSize 200
autonumber

title Sequence Diagram — полный жизненный цикл заявки в системе «Дайденег»

actor "Applicant" as A
actor "Expert" as E
actor "Holder" as H
participant "Scheduler" as S
participant "ApplicationRepository" as AppRepo
participant "ExpertRepository" as ExpRepo
participant "AssignmentRepository" as AsgRepo
participant "Distribution" as Dist
participant "ScoreEvaluator" as SE
participant "TopSelector" as TS
participant "Application" as App
participant "Assignment" as Asg
participant "Report" as Rep

== 1. Подача заявки ==

A -> App : new Application(description, area)
activate App
App --> A : app
deactivate App

A -> AppRepo : save(app)
activate AppRepo
AppRepo --> A : ok
deactivate AppRepo
note right of AppRepo
  app.status = SUBMITTED
end note

== 2. Распределение заявок (по дедлайну) ==

S -> S : runIfDeadlineReached()
note right of S : deadline наступил → true

S -> AppRepo : find_by_status(SUBMITTED)
activate AppRepo
AppRepo --> S : List<Application> pending
deactivate AppRepo

group для каждой области (area)
  S -> ExpRepo : find_by_area(area)
  activate ExpRepo
  ExpRepo --> S : List<Expert> experts
  deactivate ExpRepo

  S -> Dist : distribute(area, apps, experts)
  activate Dist

  alt experts.size() < 3
    Dist --> S : [] (warnings)
  else experts.size() >= 3
    Dist -> Dist : sort experts by load
    loop для каждой заявки
      Dist -> Dist : выбрать 3 эксперта с min load
      Dist -> Dist : result += (expert, app)
    end
    Dist --> S : List<(Expert, Application)>
  end
  deactivate Dist

  loop для каждой пары (expert, app)
    S -> Asg : new Assignment(expert.id, app.id)
    activate Asg
    Asg --> S : assignment
    deactivate Asg
    S -> App : set_status(DISTRIBUTED)
    activate App
    App --> S : ok
    deactivate App
  end

  S -> AsgRepo : save_all(assignments)
  activate AsgRepo
  AsgRepo --> S : ok
  deactivate AsgRepo
end

== 3. Оценка заявки экспертами ==

loop для каждого эксперта области
  E -> AsgRepo : find_by_expert(expert.id)
  activate AsgRepo
  AsgRepo --> E : List<Assignment> (status=ASSIGNED)
  deactivate AsgRepo

  loop для каждой заявки из Assignment
    E -> App : (проверка status == DISTRIBUTED)
    activate App
    App --> E : ok / отказ
    deactivate App

    alt status == DISTRIBUTED
      E -> App : evaluate(app, value, comment)
      activate App
      App -> App : scores.append(Score(value))
      App --> E : Score
      deactivate App

      E -> Asg : set_status(SCORED)
      activate Asg
      Asg --> E : ok
      deactivate Asg

      E -> AsgRepo : save(assignment)
      activate AsgRepo
      AsgRepo --> E : ok
      deactivate AsgRepo

      E -> AppRepo : count scores for app
      activate AppRepo
      AppRepo --> E : count
      deactivate AppRepo

      alt count == 3
        E -> SE : is_passed_threshold(app, 6, 2)
        activate SE
        SE --> E : bool
        deactivate SE

        E -> SE : calculate_final_score(app)
        activate SE
        SE --> E : double
        deactivate SE

        E -> App : set_status(PASSED_THRESHOLD / REJECTED)
        activate App
        App -> App : final_score = avg
        App --> E : ok
        deactivate App

        E -> AppRepo : save(app)
        activate AppRepo
        AppRepo --> E : ok
        deactivate AppRepo
      else count < 3
        note right of E : ожидаем остальных оценок
      end
    else status != DISTRIBUTED
      E -> Asg : set_status(EXPIRED)
      activate Asg
      Asg --> E : ok
      deactivate Asg
      note right of E : пересчёт запрещён
    end
  end
end

== 4. Отбор топ-10 по областям ==

S -> AppRepo : find_all()
activate AppRepo
AppRepo --> S : List<Application>
deactivate AppRepo

S -> TS : select_top_n_by_area(apps, 10)
activate TS
TS -> TS : group by area
loop для каждой области
  TS -> TS : sort by final_score desc
  TS -> TS : взять топ-10\n(при равенстве — все)
end
TS --> S : Map<area, List<Application>>
deactivate TS

loop для каждой заявки из топа
  S -> App : set_status(IN_TOP_N)
  activate App
  App --> S : ok
  deactivate App
  S -> AppRepo : save(app)
  activate AppRepo
  AppRepo --> S : ok
  deactivate AppRepo
end

== 5. Решение держателя ==

H -> AppRepo : find_by_status(IN_TOP_N)
activate AppRepo
AppRepo --> H : List<Application>
deactivate AppRepo

loop для каждой заявки
  alt держатель одобряет
    H -> App : set_status(APPROVED)
    activate App
    App --> H : ok
    deactivate App
  else держатель отклоняет
    H -> App : set_status(REJECTED)
    activate App
    App --> H : ok
    deactivate App
  end
  H -> AppRepo : save(app)
  activate AppRepo
  AppRepo --> H : ok
  deactivate AppRepo
end

== 6. Прикрепление отчёта победителем ==

A -> AppRepo : find_by_status(APPROVED)
activate AppRepo
AppRepo --> A : List<Application>
deactivate AppRepo

loop для каждой одобренной заявки
  A -> App : (проверка status == APPROVED)
  activate App
  App --> A : ok / отказ
  deactivate App

  alt status == APPROVED
    A -> Rep : new Report(content)
    activate Rep
    Rep --> A : report
    deactivate Rep

    A -> App : set_report(report)
    activate App
    App -> App : report_ = report
    App --> A : ok
    deactivate App

    A -> App : set_status(REPORT_SUBMITTED)
    activate App
    App --> A : ok
    deactivate App

    A -> AppRepo : save(app)
    activate AppRepo
    AppRepo --> A : ok
    deactivate AppRepo
  else status != APPROVED
    note right of A : отчёт нельзя прикрепить
  end
end

== 7. Держатель читает отчёт ==

H -> AppRepo : find_by_status(REPORT_SUBMITTED)
activate AppRepo
AppRepo --> H : List<Application>
deactivate AppRepo

loop для каждой заявки с отчётом
  H -> App : report()
  activate App
  App --> H : Report*
  deactivate App

  alt report != null && report.status == SUBMITTED
    H -> Rep : mark_as_read()
    activate Rep
    Rep -> Rep : status = READ
    Rep --> H : ok
    deactivate Rep

    H -> App : set_status(REPORT_READ)
    activate App
    App --> H : ok
    deactivate App

    H -> AppRepo : save(app)
    activate AppRepo
    AppRepo --> H : ok
    deactivate AppRepo
  else уже прочитан или отчёта нет
    note right of H : пропуск
  end
end

== 8. Финал ==

note over A, H
  Заявка в статусе REPORT_READ.
  Жизненный цикл завершён.
  Все объекты — в ApplicationRepository.
end note
